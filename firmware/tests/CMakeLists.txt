cmake_minimum_required(VERSION 3.16)
project(nomi_model_tests LANGUAGES C CXX)
enable_testing()
add_executable(face_model_test face_model_test.cpp ../main/face_model.cpp)
target_include_directories(face_model_test PRIVATE ../main)
target_compile_features(face_model_test PRIVATE cxx_std_17)
target_compile_options(face_model_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME face_model COMMAND face_model_test)
option(NOMI_RENDER_TESTS "Build real LVGL offscreen rendering checks" OFF)
if(NOMI_RENDER_TESTS)
    set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
    set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
    set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
    add_subdirectory(../components/lvgl lvgl)
    add_executable(face_render_test face_render_test.cpp ../main/face_model.cpp ../main/face_view.cpp)
    target_include_directories(face_render_test PRIVATE ../main)
    target_compile_features(face_render_test PRIVATE cxx_std_17)
    target_link_libraries(face_render_test PRIVATE lvgl::lvgl)
    add_test(NAME face_render COMMAND face_render_test)
endif()
